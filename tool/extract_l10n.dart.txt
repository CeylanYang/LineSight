import 'dart:convert';
import 'dart:io';
import 'package:analyzer/dart/analysis/utilities.dart';
import 'package:analyzer/dart/ast/ast.dart';
import 'package:analyzer/dart/ast/visitor.dart';

class Strings extends RecursiveAstVisitor<void> {
  final List<Map<String, Object>> items = [];
  String file = '';
  void collect(StringLiteral node) {
    if (node.parent is AdjacentStrings || node.parent is InterpolationExpression) return;
    String text;
    final args = <String>[];
    if (node is SimpleStringLiteral) {
      text = node.value;
    } else if (node is StringInterpolation) {
      text = '';
      for (final part in node.elements) {
        if (part is InterpolationString) text += part.value;
        if (part is InterpolationExpression) {
          text += '{${args.length}}';
          args.add(part.expression.toSource());
        }
      }
    } else { return; }
    if (!RegExp(r'[\u3400-\u9fff]').hasMatch(text)) return;
    items.add({'file': file, 'offset': node.offset, 'length': node.length, 'zh': text, 'args': args});
  }
  @override
  void visitSimpleStringLiteral(SimpleStringLiteral node) { collect(node); super.visitSimpleStringLiteral(node); }
  @override
  void visitStringInterpolation(StringInterpolation node) { collect(node); super.visitStringInterpolation(node); }
}

void main() {
  final collector = Strings();
  for (final file in Directory('lib').listSync(recursive: true).whereType<File>().where((f) => f.path.endsWith('.dart'))) {
    collector.file = file.path.replaceAll('\\', '/');
    parseString(content: file.readAsStringSync()).unit.accept(collector);
  }
  File('tool/l10n_occurrences.json').writeAsStringSync(const JsonEncoder.withIndent('  ').convert(collector.items));
  final unique = collector.items.map((x) => x['zh'] as String).toSet().toList();
  File('tool/l10n_source.json').writeAsStringSync(const JsonEncoder.withIndent('  ').convert({for (var i=0; i<unique.length; i++) '$i': unique[i]}));
  print('${collector.items.length} occurrences, ${unique.length} messages');
}
